-- !strict
-- UIEffects -> easy hover/click effects for UI
-- Every effect function returns a cleanup method
local TweenService = game:GetService("TweenService")
local UIEffects = {}
------- Variable type for config
export type EffectConfig = {
duration: number?,
easingStyle: Enum.EasingStyle?,
easingDirection: Enum.EasingDirection?,
}
----------- CONFIG --------
local DEFAULT_DURATION = 0.15
local DEFAULT_STYLE = Enum.EasingStyle.Quad
local DEFAULT_DIRECTION = Enum.EasingDirection.Out
--------------------------------
local function buildInfo(config: EffectConfig?): TweenInfo
local c = config or {}
return TweenInfo.new(
c.duration or DEFAULT_DURATION,
c.easingStyle or DEFAULT_STYLE,
c.easingDirection or DEFAULT_DIRECTION
)
end
-- Array for tracking current tweens, this helps prevent overlapping
local activeTweens: { [GuiObject]: Tween } = {}
local function playTween(element: GuiObject, info: TweenInfo, goal: { [string]: any })
local existing = activeTweens[element]
if existing then
existing:Cancel()
end
local tween = TweenService:Create(element, info, goal)
activeTweens[element] = tween
tween:Play()
end
-- Builder for function that disposing of connections
local function bindHover(element: GuiObject, onEnter: () -> (), onLeave: () -> ())
local enterConn = element.MouseEnter:Connect(onEnter)
local leaveConn = element.MouseLeave:Connect(onLeave)
local destroyConn = element.Destroying:Connect(function()
activeTweens[element] = nil
end)
return function()
enterConn:Disconnect()
leaveConn:Disconnect()
destroyConn:Disconnect()
activeTweens[element] = nil
end
end
--- Tween BackgroundColor3 on hover
function UIEffects.colorChange(
element: GuiObject,
hoverColor: Color3,
config: EffectConfig?
)
local info = buildInfo(config)
local original = element.BackgroundColor3
return bindHover(element,
function() playTween(element, info, { BackgroundColor3 = hoverColor }) end,
function() playTween(element, info, { BackgroundColor3 = original }) end
)
end
--- Scale the element on hover, scales both Scale and Offset of ui
function UIEffects.scale(
element: GuiObject,
factor: number,
config: EffectConfig?
)
local info = buildInfo(config)
local original = element.Size
local hovered = UDim2.new(
original.X.Scale * factor, original.X.Offset * factor,
original.Y.Scale * factor, original.Y.Offset * factor
)
return bindHover(element,
function() playTween(element, info, { Size = hovered }) end,
function() playTween(element, info, { Size = original }) end
)
end
--- Fade Background Transparency on hover
function UIEffects.fade(
element: GuiObject,
hoverTransparency: number,
config: EffectConfig?
)
local info = buildInfo(config)
local original = element.BackgroundTransparency
return bindHover(element,
function() playTween(element, info, { BackgroundTransparency = hoverTransparency }) end,
function() playTween(element, info, { BackgroundTransparency = original }) end
)
end
return UIEffects
--[[
SkinsShop
Handles skins - crates purchase and equipping
--]]
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerStorage = game:GetService("ServerStorage")
------------------- SERVICES --------
local Events = ReplicatedStorage:WaitForChild("Events")
local SkinsBuyEvent = Events:WaitForChild("SkinsBuy") :: RemoteEvent
local SkinEquipEvent = Events:WaitForChild("SkinEquip") :: RemoteEvent
local AnnounceEvent = Events:WaitForChild("AnnouncePlayer") :: RemoteEvent
local Modules = ReplicatedStorage:WaitForChild("Modules")
local RollModule = require(Modules:WaitForChild("RollModule"))
local SkinPools = require(Modules:WaitForChild("SkinsPool"))
local DataManager = require(ServerStorage.Data.DataManager)
------------------- CONFIG -----------
local CRATE_PRICES = {
BaseMirror = 1000,
}
local MIN_ROLL_AMOUNT = 1
local MAX_ROLL_AMOUNT = 1000
local PURCHASE_COOLDOWN_SECONDS = 0.5 -- This way exploiters cant open 1000 crates at once
local ANNOUNCE_DURATION_SECONDS = 2
---------- PLAYERS COOLDOWN
local lastPurchaseAt = {}
Players.PlayerRemoving:Connect(function(player)
lastPurchaseAt[player] = nil
end)
---------------- Helping Functions
local function notify(player, message)
AnnounceEvent:FireClient(player, message, ANNOUNCE_DURATION_SECONDS)
end
local function sanitizeAmount(raw)
-- rejects invalid input
if typeof(raw) ~= "number" or raw ~= raw then
return MIN_ROLL_AMOUNT
end
return math.clamp(math.floor(raw), MIN_ROLL_AMOUNT, MAX_ROLL_AMOUNT)
end
local function isOnCooldown(player)
local last = lastPurchaseAt[player]
return last ~= nil and (os.clock() - last) < PURCHASE_COOLDOWN_SECONDS
end
---------- HANDLING PLAYER EQUIPPING
SkinEquipEvent.OnServerEvent:Connect(function(player, skinName)
if typeof(skinName) ~= "string" then
return
end
local profile = DataManager.Profiles[player]
if not profile then
return
end
if not table.find(profile.Data.OwnedSkins, skinName) then
notify(player, "You don't own this skin.")
return
end
DataManager:Set(player, "EquippedSkin", skinName)
end)
------------- ROLLING FUNCTION
local function buildRewards(pool, rolls, ownedSkins)
local rewards = {}
for _ = 1, rolls do
local rolled = RollModule.Roll(pool)
local entry = rewards[rolled]
if not entry then
entry = {
count = 0,
isNew = table.find(ownedSkins, rolled) == nil,
}
rewards[rolled] = entry
end
entry.count += 1
end
return rewards
end
-------- HANDLING CRATES PURCHASE
SkinsBuyEvent.OnServerEvent:Connect(function(player, skinName, amount)
------------------ 1 - GUARD CHECKS
if typeof(skinName) ~= "string" then
return
end
if isOnCooldown(player) then
return
end
lastPurchaseAt[player] = os.clock()
local pool = SkinPools[skinName]
local unitPrice = CRATE_PRICES[skinName]
if not pool or not unitPrice then
return
end
local profile = DataManager.Profiles[player]
if not profile then
return
end
------------------ 2 - CHECKING PRICE
local rollCount = sanitizeAmount(amount)
local totalCost = unitPrice * rollCount
if profile.Data.Ahn < totalCost then
notify(player, "Not enough Ahn!")
return
end
------------------------- 3 - APPLY CHANGES
DataManager:Set(player, "Ahn", profile.Data.Ahn - totalCost)
local rewards = buildRewards(pool, rollCount, profile.Data.OwnedSkins)
local ownedSkins = profile.Data.OwnedSkins
for name, entry in rewards do
if entry.isNew then
table.insert(ownedSkins, name)
end
end
DataManager:Set(player, "OwnedSkins", ownedSkins)
------------------------- 4 - NOTIFY CLIENT (FOR VISUALS)
SkinsBuyEvent:FireClient(player, rewards)
end)

RoundManager
Server-side controller for round-based multiplayer game.
--]]
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
-------------- SERVICES -------
local Modules = ReplicatedStorage:WaitForChild("Modules")
local PlayerManager = require(Modules:WaitForChild("PlayerManager"))
local ArenaService = require(Modules:WaitForChild("ArenaService"))
local Config = require(script:WaitForChild("Config"))
---------------- STATIC
local Phases = {
WAITING = "Waiting",
INTERMISSION = "Intermission",
ROUND = "Round",
ROUND_END = "RoundEnd",
}
-------------- HELPERS
local function setPhase(phase, data)
currentPhase = phase
PhaseChanged:FireAllClients(phase, data or {})
end
local function countdown(seconds: number, label: string)
for remaining = seconds, 0, -1 do
RoundInfo:FireAllClients(label, remaining)
task.wait(1)
end
end
----------- ROUND MANAGEMENT
local function runRound()
local participants = PlayerManager.GetAllPlayers()
setPhase(Phases.ROUND, { playerCount = #participants })
for _, player in participants do
player:LoadCharacterAsync()
end
ArenaService.SpawnInCircle(participants)
countdown(Config.ROUND_TIME, "Round")
end
-------------- MAIN LOOP
local function gameLoop()
while true do
waitForPlayers()
runIntermission()
runRound()
endRound()
end
end
task.spawn(gameLoop)
